cmake_minimum_required(VERSION 3.20)
project(VRDemoChecks LANGUAGES CXX)
find_package(SaidaEngine CONFIG REQUIRED)
add_executable(vrdemo_check Check.cpp)
target_link_libraries(vrdemo_check PRIVATE Saida::Engine)
get_filename_component(VRDEMO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
target_compile_definitions(vrdemo_check PRIVATE VRDEMO_ROOT="${VRDEMO_ROOT}")
enable_testing()
add_test(NAME vrdemo_headless COMMAND vrdemo_check)
