cmake_minimum_required(VERSION 3.20)
project(SaidaPortableMedia LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(SAIDA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
add_library(saida_media STATIC "${SAIDA_ROOT}/src/audio/AudioUsage.cpp")
target_include_directories(saida_media PRIVATE "${SAIDA_ROOT}/third_party/miniaudio" "${SAIDA_ROOT}/third_party/stb")
include("${SAIDA_ROOT}/cmake/MediaRuntime.cmake")
saida_add_media_runtime(saida_media)
if(EMSCRIPTEN)
    add_executable(media_web "${SAIDA_ROOT}/tests/portable_media/WebFixture.cpp")
    target_link_libraries(media_web PRIVATE saida_media)
    target_compile_options(saida_media PRIVATE -fexceptions)
    target_link_options(media_web PRIVATE -sALLOW_MEMORY_GROWTH=1 -sEXIT_RUNTIME=0 -sDISABLE_EXCEPTION_CATCHING=0 -sEXPORTED_RUNTIME_METHODS=ccall -sEXPORTED_FUNCTIONS=_main)
elseif(ANDROID)
    add_library(media_android_link SHARED MobileLinkFixture.cpp)
    target_link_libraries(media_android_link PRIVATE saida_media)
    target_link_options(media_android_link PRIVATE -Wl,--no-undefined)
else()
    add_executable(media_test "${SAIDA_ROOT}/tests/AudioStreamTests.cpp")
    target_link_libraries(media_test PRIVATE saida_media)
    add_executable(http_test "${SAIDA_ROOT}/tests/HttpRequestTests.cpp")
    target_link_libraries(http_test PRIVATE saida_media)
endif()
